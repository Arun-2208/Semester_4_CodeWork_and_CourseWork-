> # Helper: robust parse for accident_date
> parse_accident_date <- function(x_chr){
+   # Try many common formats (with/without time)
+   dt <- suppressWarnings(parse_date_time(
+     x_chr,
+     orders = c(
+       "dmy","ymd","mdy",
+       "dmy HMS","ymd HMS","mdy HMS",
+       "d/m/Y","Y/m/d","m/d/Y",
+       "d-m-Y","Y-m-d","m-d-Y",
+       "d.m.Y","Y.m.d",
+       "d%b%Y","b%d%Y"      # e.g., 15Jan2018, Jan15 2018
+     ),
+     tz = "UTC"
+   ))
+ 
+   # If some entries look numeric (Excel serials), convert those
+   x_num <- suppressWarnings(as.numeric(x_chr))
+   needs_excel <- is.na(dt) & !is.na(x_num)
+   if(any(needs_excel)){
+     dt[needs_excel] <- as.Date(x_num[needs_excel], origin = "1899-12-30")
+   }
+ 
+   # Return pure Date
+   as.Date(dt)
+ }

Step 5 — Speed environment across all periods

goal: for each period × Urban/Rural × speed bucket (posted limits), compute:

severity: fatalities per 1,000 crashes

exposure: share of crashes in each bucket

then make:

a faceted bar chart (per period) for severity, and

a faceted scatter (per period) with x = exposure share and y = severity, coloured by Urban/Rural and shaped by speed bucket.

assumes you already have acc_p from earlier steps.


> # Re-parse with a very permissive approach & Excel serial fallback
> reparse_date <- function(x_chr){
+     x_chr <- as.character(x_chr)
+     dt <- suppressWarnings(lubridate::parse_date_time(
+         x_chr,
+         orders = c(
+             "Y-m-d","d-m-Y","m-d-Y","Y/m/d","d/m/Y","m/d/Y",
+             "d.m.Y","Y.m.d","Ymd","dmY","mdY",
+             "Y-m-d HMS","d-m-Y HMS","m-d-Y HMS",
+             "Y/m/d HMS","d/m/Y HMS","m/d/Y HMS"
+         ), tz = "UTC"
+     ))
+     # Excel serials?
+     num <- suppressWarnings(as.numeric(x_chr))
+     need_serial <- is.na(dt) & !is.na(num)
+     if(any(need_serial)) dt[need_serial] <- as.Date(num[need_serial], origin = "1899-12-30")
+     as.Date(dt)
+ }

Step 6 — Road geometry across all periods


6.1 Map raw geometry descriptions → stable groups (robust)

We handle common variants across years like “NO INTERSECTION”, “MID-BLOCK”, “T-JUNCTION”, “CROSS ROAD”, etc.

> period_order <- c("2012–2014","2015–2017","2018–2020","2021–2023","2024–2025")
> 
> acc_geom <- acc_p %>%
+     mutate(
+         geom_raw = toupper(trimws(as.character(road_geometry_desc))),
+         geom_group = case_when(
+             # Open-road / non-intersection variants
+             !is.na(geom_raw) & grepl("NOT\\s*AT\\s*INTERSECTION|NO\\s*INTERSECTION|MID[- ]?BLOCK|OPEN\\s*ROAD|NON[- ]?INTERSECTION|MIDBLOCK|MID BLOCK", geom_raw)
+             ~ "Not at intersection",
+             # T intersection variants
+             !is.na(geom_raw) & grepl("\\bT[- ]?INTERSECTION\\b|\\bT[- ]?INT\\b|\\bT[- ]?JUNCTION\\b|\\bT JN\\b", geom_raw)
+             ~ "T intersection",
+             # Cross/4-way variants
+             !is.na(geom_raw) & grepl("CROSS( |-)?INTERSECTION|CROSS ?ROAD(S)?|FOUR[- ]?WAY|4[- ]?WAY|X[- ]?ROAD|CROSSING\\b", geom_raw)
+             ~ "Cross intersection",
+             # Roundabout variants
+             !is.na(geom_raw) & grepl("ROUNDABOUT|R'?ABOUT|RAB", geom_raw)
+             ~ "Roundabout",
+             TRUE ~ "Other/Unknown"
+         ),
+         period = factor(period, levels = period_order),
+         urban_rural = if_else(urban_rural %in% c("URBAN","RURAL"), urban_rural, "UNKNOWN")
+     )
>
