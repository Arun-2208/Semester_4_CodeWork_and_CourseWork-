# 1) Helpers ----------------------------------------------------------
parse_accident_date <- function(x_chr){
  x_chr <- as.character(x_chr)
  dt <- suppressWarnings(lubridate::parse_date_time(
    x_chr,
    orders = c(
      "Y-m-d","d-m-Y","m-d-Y","Y/m/d","d/m/Y","m/d/Y",
      "d.m.Y","Y.m.d","Ymd","dmY","mdY",
      "Y-m-d HMS","d-m-Y HMS","m-d-Y HMS",
      "Y/m/d HMS","d/m/Y HMS","m/d/Y HMS"
    ),
    tz = "UTC"
  ))
  num <- suppressWarnings(as.numeric(x_chr))              # Excel serials
  need_serial <- is.na(dt) & !is.na(num)
  if(any(need_serial)) dt[need_serial] <- as.Date(num[need_serial], origin = "1899-12-30")
  as.Date(dt)
}

label_period <- function(y){
  dplyr::case_when(
    y %in% 2012:2014 ~ "2012–2014",
    y %in% 2015:2017 ~ "2015–2017",
    y %in% 2018:2020 ~ "2018–2020",
    y %in% 2021:2023 ~ "2021–2023",
    y %in% 2024:2025 ~ "2024–2025",
    TRUE ~ NA_character_
  )
}
years_in_period <- c("2012–2014"=3,"2015–2017"=3,"2018–2020"=3,"2021–2023"=3,"2024–2025"=2)
period_order    <- names(years_in_period)

# 2) Load data (CSV files in /data) ----------------------------------
acc_path  <- "data/accident.csv"
node_path <- "data/node.csv"

stopifnot(file.exists(acc_path), file.exists(node_path))

acc  <- readr::read_csv(acc_path,  show_col_types = FALSE) %>% janitor::clean_names()
node <- readr::read_csv(node_path, show_col_types = FALSE) %>% janitor::clean_names()

# Basic column checks (fail fast with a clear message)
needed_acc_cols  <- c("accident_no","accident_date","no_persons_killed")
missing_acc      <- setdiff(needed_acc_cols, names(acc))
if(length(missing_acc)) stop("Missing in accident.csv: ", paste(missing_acc, collapse=", "))

needed_node_cols <- c("accident_no","deg_urban_name")
missing_node     <- setdiff(needed_node_cols, names(node))
if(length(missing_node)) stop("Missing in node.csv: ", paste(missing_node, collapse=", "))

# 3) Build the Geo4 label and join -----------------------------------
geo4_keep <- c("MELB_URBAN","RURAL_VICTORIA","SMALL_CITIES","TOWNS")

node_geo4 <- node %>%
  mutate(
    deg_urban_name = toupper(trimws(as.character(deg_urban_name))),
    geo4 = if_else(deg_urban_name %in% geo4_keep, deg_urban_name, NA_character_)
  ) %>%
  select(accident_no, geo4)

acc4 <- acc %>%
  mutate(
    accident_date     = parse_accident_date(accident_date),
    year              = lubridate::year(accident_date),
    no_persons_killed = suppressWarnings(as.numeric(no_persons_killed))
  ) %>%
  left_join(node_geo4, by = "accident_no") %>%
  mutate(period = label_period(year)) %>%
  filter(!is.na(period), !is.na(geo4)) %>%
  mutate(
    period = factor(period, levels = period_order),
    geo4   = factor(geo4,   levels = geo4_keep)
  )

# 4) Baseline severity per 1,000 by period × Geo4 --------------------
baseline_geo4 <- acc4 %>%
  group_by(period, geo4) %>%
  summarise(
    crashes    = n(),
    fatalities = sum(coalesce(no_persons_killed, 0), na.rm = TRUE),
    .groups    = "drop"
  ) %>%
  mutate(
    years            = years_in_period[as.character(period)],
    crashes_per_year = crashes / years,
    fatal_per_1000   = 1000 * fatalities / crashes
  ) %>%
  arrange(period, geo4)

# Save table + plot
readr::write_csv(baseline_geo4, "output/insight_baseline_geo4.csv")

p_base4 <- ggplot(baseline_geo4,
                  aes(x = period, y = fatal_per_1000, color = geo4, group = geo4)) +
  geom_line(linewidth = 1) +
  geom_point(size = 3) +
  labs(
    title = "Fatalities per 1,000 crashes — MELB_URBAN vs RURAL_VICTORIA vs SMALL_CITIES vs TOWNS",
    x = NULL, y = "Fatalities per 1,000 crashes", color = "Geography"
  ) +
  theme_minimal(base_size = 13)

ggsave("output/insight_baseline_geo4.png", p_base4, width = 12, height = 6, dpi = 150)

message("Baseline complete. Files written to /output")

  mutate(
    speed_numeric = readr::parse_number(as.character(speed_zone)),
    speed_bucket = case_when(
      speed_numeric %in% c(10,20,30,40,50,60) ~ "<=60",

  filter(speed_bucket != "Unknown") %>%
  group_by(period, geo4, speed_bucket) %>%

    crashes = n(),

    fatal_per_1000 = 1000 * fatalities / crashes,
    .groups = "drop"
  )

speed_exp_geo4 <- acc4_speed %>%
  filter(speed_bucket != "Unknown") %>%
  count(period, geo4, speed_bucket, name = "crashes") %>%
  group_by(period, geo4) %>%
  mutate(share = crashes / sum(crashes), share_pct = 100 * share) %>%
  ungroup()
readr::write_csv(speed_exp_geo4, "output/insight_speed_exposure_all_periods_geo4.csv")

  speed_sev_geo4,
  speed_exp_geo4 %>% select(period, geo4, speed_bucket, share, share_pct),
  by = c("period","geo4","speed_bucket")
) %>%
  mutate(period = factor(period, levels = period_order))

# --- Normalise lighting labels ---
normalize_light <- function(x){
  x_chr <- toupper(trimws(as.character(x)))
  x_num <- suppressWarnings(readr::parse_number(x_chr))
  dplyr::case_when(
    !is.na(x_num) & x_num == 1 ~ "Daylight",
    !is.na(x_num) & x_num == 2 ~ "Dusk/Dawn",
    !is.na(x_num) & x_num == 3 ~ "Dark—Lighted",
    !is.na(x_num) & x_num == 4 ~ "Dark—Unlighted",
    grepl("\\bDAY(LIGHT)?\\b", x_chr)                              ~ "Daylight",
    grepl("\\bDUSK|DAWN|TWILIGHT\\b", x_chr)                       ~ "Dusk/Dawn",
    grepl("DARK.*(LIT|LIGHTED|STREET LIGHT|LAMP|ILLUMIN)", x_chr)  ~ "Dark—Lighted",
    grepl("DARK.*(UNLIT|UNLIGHTED|NO LIGHT|NO STREET)", x_chr)     ~ "Dark—Unlighted",
    TRUE ~ "Other/Unknown"
  )
}
